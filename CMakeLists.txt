cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(BATTranspose VERSION 0.2.0 LANGUAGES C CXX)
if(APPLE)
  # Enable at the common parent before the SDK creates any Objective-C++ targets.
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BAT_BUILD_PLUGIN "Build the Windows, Linux or macOS VST3" ON)
option(BAT_BUILD_TESTS "Build DSP and plugin tests" ON)
set(BAT_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/external" CACHE PATH "Dependency folder")
add_library(bat_dsp STATIC src/dsp/TransposeEngine.cpp src/dsp/CorrelationShifter.cpp)
target_include_directories(bat_dsp PUBLIC src)
if(MSVC)
  target_compile_options(bat_dsp PRIVATE /O2 /bigobj /W4)
endif()
if(BAT_BUILD_TESTS)
  enable_testing()
  add_executable(bat_dsp_tests tests/dsp_tests.cpp)
  target_link_libraries(bat_dsp_tests PRIVATE bat_dsp)
  add_test(NAME dsp COMMAND bat_dsp_tests)
endif()
if(BAT_BUILD_PLUGIN)
  if(NOT WIN32 AND NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Supported plugin platforms: Windows, Linux and macOS")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "BAT-transpose requires a 64-bit toolchain")
  endif()
  if(NOT EXISTS "${BAT_DEPS}/vst3sdk/vstgui4/vstgui/lib/cframe.h")
    message(FATAL_ERROR "VSTGUI is missing; rerun the bootstrap script")
  endif()
  set(SMTG_ENABLE_VSTGUI_SUPPORT ON CACHE BOOL "" FORCE)
  set(VSTGUI_ENABLE_OPENGL_SUPPORT OFF CACHE BOOL "" FORCE)
  set(VSTGUI_ENABLE_XMLPARSER OFF CACHE BOOL "" FORCE)
  set(VSTGUI_UISCRIPTING OFF CACHE BOOL "" FORCE)
  # X11 embedding (also via XWayland); native Wayland is not part of this release.
  set(VSTGUI_ENABLE_WAYLAND_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
  add_subdirectory("${BAT_DEPS}/vst3sdk" vst3sdk EXCLUDE_FROM_ALL)
  set(SDK_ROOT "${BAT_DEPS}/vst3sdk")
  set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
  smtg_add_vst3plugin(BATTranspose PACKAGE_NAME "BAT-transpose" SOURCES_LIST
    src/plugin/Processor.cpp src/plugin/Controller.cpp src/plugin/Editor.cpp src/plugin/Factory.cpp)
  target_link_libraries(BATTranspose PRIVATE bat_dsp sdk vstgui_support)
  target_include_directories(BATTranspose PRIVATE src)
  target_compile_definitions(BATTranspose PRIVATE BAT_VERSION="${PROJECT_VERSION}")
  if(WIN32)
    target_compile_definitions(BATTranspose PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  elseif(APPLE)
    smtg_target_set_bundle(BATTranspose BUNDLE_IDENTIFIER "audio.bat.transpose" COMPANY_NAME "BAT")
  endif()
  # SDK 3.8 sets per-configuration bundle directories only for multi-config generators.
  if(WIN32 AND NOT CMAKE_CONFIGURATION_TYPES)
    set_target_properties(BATTranspose PROPERTIES LIBRARY_OUTPUT_DIRECTORY
      "${CMAKE_BINARY_DIR}/VST3/${CMAKE_BUILD_TYPE}/BAT-transpose.vst3/Contents/x86_64-win")
  endif()
  get_target_property(BAT_BUNDLE BATTranspose SMTG_PLUGIN_PACKAGE_PATH)
  install(DIRECTORY "${BAT_BUNDLE}" DESTINATION .)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION BAT-transpose.vst3/Contents/Resources)
  install(FILES README.md screenshot.png LICENSE DESTINATION .)

  if(APPLE)
    # Resources added during staging must be covered by the local ad-hoc signature.
    install(CODE [[
      execute_process(COMMAND /usr/bin/codesign --force --sign -
        "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/BAT-transpose.vst3"
        COMMAND_ERROR_IS_FATAL ANY)
    ]])
    if(CMAKE_OSX_ARCHITECTURES MATCHES "arm64" AND CMAKE_OSX_ARCHITECTURES MATCHES "x86_64")
      set(BAT_PACKAGE_PLATFORM macos-universal)
    else()
      set(BAT_PACKAGE_PLATFORM "macos-${CMAKE_OSX_ARCHITECTURES}")
    endif()
  elseif(WIN32)
    set(BAT_PACKAGE_PLATFORM windows-x64)
  else()
    set(BAT_PACKAGE_PLATFORM "linux-${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  set(CPACK_GENERATOR ZIP)
  set(CPACK_PACKAGE_NAME BAT-transpose)
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_PACKAGE_FILE_NAME "BAT-transpose-${PROJECT_VERSION}-${BAT_PACKAGE_PLATFORM}")
  set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
  include(CPack)
  if(BAT_BUILD_TESTS)
    add_custom_target(bat_validate
      COMMAND "$<TARGET_FILE:validator>" "${BAT_BUNDLE}"
      DEPENDS BATTranspose validator
      USES_TERMINAL VERBATIM)
    if(WIN32)
      set(BAT_HOST_MODULE module_win32.cpp)
    elseif(APPLE)
      set(BAT_HOST_MODULE module_mac.mm)
    else()
      set(BAT_HOST_MODULE module_linux.cpp)
    endif()
    add_executable(bat_host tools/vst_host.cpp
      "${BAT_DEPS}/vst3sdk/public.sdk/source/common/memorystream.cpp"
      "${BAT_DEPS}/vst3sdk/public.sdk/source/vst/hosting/${BAT_HOST_MODULE}"
      "${BAT_DEPS}/vst3sdk/public.sdk/source/vst/hosting/plugprovider.cpp")
    target_link_libraries(bat_host PRIVATE sdk_hosting ${CMAKE_DL_LIBS})
    if(WIN32)
      target_link_libraries(bat_host PRIVATE ole32 user32 gdi32)
      target_compile_definitions(bat_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    elseif(APPLE)
      target_link_libraries(bat_host PRIVATE "-framework Cocoa")
      target_compile_options(bat_host PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    endif()
    target_include_directories(bat_host PRIVATE tools)
    add_test(NAME plugin_state COMMAND bat_host "${BAT_BUNDLE}" --state-test)
    if(WIN32)
      add_test(NAME plugin_editor COMMAND bat_host "${BAT_BUNDLE}" --editor-test)
    endif()
  endif()
endif()
